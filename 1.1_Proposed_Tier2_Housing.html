Proposed Tier 2 Housing NH3 Emission Factors

Overview

This file calculates proposed Tier 2 housing ammonia emission factors by animal type, housing type, manure type, and broad climate zone.

The deterministic output uses broad climate-zone outdoor annual temperature, converted to an animal-specific indoor housing temperature, and the updated housing correction-factor table:

EF = EFs_TotalN * CT_15^(T_indoor - T_air)

EFs_TotalN is treated as the reference EF at the row-specific indoor air temperature T_air from Inputs/NH3_Correction_Factors.xlsx.

clean_climate_table <- function(climate_table) {
  climate_table$Climate_zone <- trimws(as.character(climate_table$Climate_zone))
  climate_table <- climate_table[
    !is.na(climate_table$Climate_zone) &
      nzchar(climate_table$Climate_zone) &
      climate_table$Climate_zone != "0",
    ,
    drop = FALSE
  ]

  for (month in month_labels) {
    climate_table[[month]] <- suppressWarnings(as.numeric(climate_table[[month]]))
  }

  climate_table
}

climate_medians <- clean_climate_table(climate_medians)

get_climate_matrix <- function(climate_table, component_zones) {
  climate_rows <- climate_table[climate_table$Climate_zone %in% component_zones, ]

  if (nrow(climate_rows) != length(component_zones)) {
    stop(
      "Expected climate rows for: ",
      paste(component_zones, collapse = ", ")
    )
  }

  as.matrix(climate_rows[month_labels])
}

Calculation Functions

cap_ef <- function(ef_value) {
  pmin(ef_value, ef_max)
}

housing_temperature_group <- function(animal_type) {
  animal_type <- tolower(as.character(animal_type))

  if (grepl("cattle", animal_type)) {
    return("cattle")
  }
  if (grepl("swine", animal_type)) {
    return("swine")
  }
  if (grepl("broiler", animal_type)) {
    return("broiler")
  }
  if (grepl("layer", animal_type)) {
    return("layer")
  }

  stop("No indoor temperature equation defined for animal type: ", animal_type)
}

calculate_indoor_temperature <- function(outdoor_temperature_c, animal_type) {
  group <- housing_temperature_group(animal_type)

  if (identical(group, "cattle")) {
    return(outdoor_temperature_c + 3)
  }

  if (identical(group, "swine")) {
    return(ifelse(
      outdoor_temperature_c <= 0,
      20 + 0.5 * outdoor_temperature_c,
      ifelse(
        outdoor_temperature_c <= 12.5,
        20,
        20 + (outdoor_temperature_c - 12.5)
      )
    ))
  }

  if (identical(group, "broiler")) {
    return(
      2.0e-4 * outdoor_temperature_c^3 +
        1.0e-3 * outdoor_temperature_c^2 +
        2.4e-2 * outdoor_temperature_c +
        22.1
    )
  }

  1.4e-4 * outdoor_temperature_c^3 +
    2.3e-3 * outdoor_temperature_c^2 +
    1.1e-2 * outdoor_temperature_c +
    23.8
}

# Notes such as "Only applicable to T >15" define the minimum corrected
# housing temperature required before CT is applied.
threshold_from_note <- function(note) {
  note <- as.character(note)
  match_text <- regmatches(
    note,
    regexpr(">[[:space:]]*[-+]?[0-9]*\\.?[0-9]+", note)
  )

  if (length(match_text) == 0 || is.na(match_text) || match_text == "") {
    return(NA_real_)
  }

  as.numeric(gsub(">|[[:space:]]", "", match_text))
}

ct_is_applicable <- function(ct, corrected_temperature_c, note) {
  if (is.na(ct)) {
    return(FALSE)
  }

  threshold_c <- threshold_from_note(note)
  if (is.na(threshold_c)) {
    return(TRUE)
  }

  corrected_temperature_c > threshold_c
}

calculate_annual_ef <- function(
  ef_ref,
  ct,
  corrected_temperature_c,
  t_air_reference_c,
  note
) {
  if (ct_is_applicable(ct, corrected_temperature_c, note)) {
    corrected_ef <- ef_ref * ct^(corrected_temperature_c - t_air_reference_c)
  } else {
    corrected_ef <- ef_ref
  }

  cap_ef(corrected_ef)
}

Monte Carlo Preparation

# Bounded lognormal sampling uses the reported lower and upper 95% CI limits
# as bounds. Values outside the bounds are resampled.
sample_bounded_lognormal <- function(n, center, lower, upper) {
  if (is.na(center) || is.na(lower) || is.na(upper) ||
      center <= 0 || lower <= 0 || upper <= 0 || lower >= upper) {
    return(rep(center, n))
  }

  sdlog <- (log(upper) - log(lower)) / (2 * 1.96)
  samples <- rep(NA_real_, n)
  remaining <- seq_len(n)
  attempts <- 0

  while (length(remaining) > 0 && attempts < 100) {
    draws <- rlnorm(length(remaining), meanlog = log(center), sdlog = sdlog)
    keep <- draws >= lower & draws <= upper

    if (any(keep)) {
      samples[remaining[keep]] <- draws[keep]
    }

    remaining <- remaining[!keep]
    attempts <- attempts + 1
  }

  # If rejection sampling leaves a few empty values, use the point estimate.
  samples[is.na(samples)] <- center
  samples
}

sample_truncated_normal <- function(n, mean, lower, upper) {
  if (is.na(mean) || is.na(lower) || is.na(upper) || lower >= upper) {
    return(rep(mean, n))
  }

  sd <- (upper - lower) / (2 * 1.96)
  if (is.na(sd) || sd <= 0) {
    return(rep(mean, n))
  }

  samples <- rep(NA_real_, n)
  remaining <- seq_len(n)
  attempts <- 0

  while (length(remaining) > 0 && attempts < 100) {
    draws <- rnorm(length(remaining), mean = mean, sd = sd)
    keep <- draws >= lower & draws <= upper

    if (any(keep)) {
      samples[remaining[keep]] <- draws[keep]
    }

    remaining <- remaining[!keep]
    attempts <- attempts + 1
  }

  samples[is.na(samples)] <- mean
  samples
}

outdoor_temperature_samples <- list()
for (climate_col in climate_cols) {

Monte Carlo Simulation

mc_summary_rows <- list()
mc_summary_index <- 1

for (row_index in seq_len(nrow(factors))) {
  ef_samples <- sample_bounded_lognormal(
    n = n_iter,
    center = factors$EFs_TotalN[row_index],
    lower = factors$EFs_Lower[row_index],
    upper = factors$EFs_Upper[row_index]
  )

  ct_samples <- sample_bounded_lognormal(
    n = n_iter,
    center = factors$CT_15[row_index],
    lower = factors$CT_Lower[row_index],
    upper = factors$CT_Upper[row_index]
  )

  t_air_samples <- sample_truncated_normal(
    n = n_iter,
    mean = factors$T_air[row_index],
    lower = factors$T_air_Lower[row_index],
    upper = factors$T_air_Upper[row_index]
  )

  threshold_c <- threshold_from_note(factors$Note[row_index])

  for (climate_col in climate_cols) {
    outdoor_temp_samples <- outdoor_temperature_samples[[climate_col]]
    indoor_temp_samples <- calculate_indoor_temperature(
      outdoor_temperature_c = outdoor_temp_samples,
      animal_type = factors$Animal_Type[row_index]
    )

    if (is.na(threshold_c)) {
      ct_applied <- !is.na(ct_samples)
    } else {
      ct_applied <- !is.na(ct_samples) & indoor_temp_samples > threshold_c
    }

    ef_sim <- ifelse(
      ct_applied,
      ef_samples * ct_samples^(indoor_temp_samples - t_air_samples),
      ef_samples
    )
    ef_sim <- cap_ef(ef_sim)

    deterministic_ef <- proposed_result[row_index, climate_col]

    mc_summary_rows[[mc_summary_index]] <- data.frame(
      Animal_Type = factors$Animal_Type[row_index],
      Housing = factors$Housing[row_index],
      Manure_type = factors$Manure_type[row_index],
      Climate_zone = climate_col,
      Outdoor_Annual_Temperature_C = round(
        broad_climate_summary[
          broad_climate_summary$Climate_zone == climate_col,
          "Outdoor_Annual_Temperature_C"
        ],
        3
      ),
      Outdoor_Annual_Temperature_SD_C = round(
        broad_climate_summary[
          broad_climate_summary$Climate_zone == climate_col,
          "Outdoor_Annual_Temperature_SD_C"
        ],
        3
      ),
      Indoor_Annual_Temperature_C = round(
        calculate_indoor_temperature(
          broad_climate_summary[
            broad_climate_summary$Climate_zone == climate_col,
            "Outdoor_Annual_Temperature_C"
          ],
          factors$Animal_Type[row_index]
        ),
        3
      ),
      T_air_Reference_C = factors$T_air[row_index],
      T_air_Lower_C = factors$T_air_Lower[row_index],
      T_air_Upper_C = factors$T_air_Upper[row_index],
      EF_Deterministic = round(deterministic_ef, 3),
      EF_MC_Mean = round(mean(ef_sim, na.rm = TRUE), 3),
      EF_MC_Lower95 = round(as.numeric(quantile(ef_sim, 0.025, na.rm = TRUE)), 3),
      EF_MC_Upper95 = round(as.numeric(quantile(ef_sim, 0.975, na.rm = TRUE)), 3),
      CT_Threshold_C = threshold_c,
      stringsAsFactors = FALSE
    )

    mc_summary_index <- mc_summary_index + 1
  }
}

mc_summary <- do.call(rbind, mc_summary_rows)

Method Notes

The final proposal uses annual outdoor temperature only for housing climate assignment. Monthly temperature medians are used only to construct annual broad outdoor temperatures from the climate workbook.

Broad climate zones are represented by averaging component climate-zone monthly medians before calculating the annual mean. Warm temperate and cool temperate therefore average their moist and dry component zones.

For the Monte Carlo simulation, monthly outdoor temperatures are not sampled independently. Instead, one annual outdoor temperature is sampled for each broad climate zone using the annual mean temperature and an annual-level standard deviation derived from the workbook’s monthly Std values. The SD is treated as a representative broad-zone variability term, not as a formal uncertainty of the mean climate-zone average. Each outdoor draw is then converted to animal-specific indoor temperature before EF correction.

The correction uses EFs_TotalN as the row-specific T_air reference EF. Monte Carlo simulations also sample T_air from T_air_Lower and T_air_Upper. The input column EF_Ref_15 is retained in the input table but is not used in the main proposed output.
