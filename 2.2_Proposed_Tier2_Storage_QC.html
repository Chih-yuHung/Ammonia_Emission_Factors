QC: Proposed Tier 2 Storage Monthly TAN and NH3

Overview

This QC file shows the month-by-month storage mass balance used for the proposed Tier 2 storage NH3 emission factor calculation.

The QC scenarios are:

  • Dairy Cattle, Tank, Slurry
  • Dairy Cattle, Manure heap, Solid

Both are evaluated for Cool Temperate Moist under all four removal schedules.

Calculation Functions

cap_ef <- function(ef_value) {
  pmin(ef_value, ef_max)
}

ct_basis_for_climate <- function(climate_zone) {
  if (grepl("Dry$", climate_zone)) {
    return("dry")
  }

  "wet"
}

calculate_monthly_ef <- function(ef_ref, ct, temperatures_c) {
  if (is.na(ct)) {
    return(cap_ef(rep(ef_ref, length(temperatures_c))))
  }

  cap_ef(ef_ref * ct^(temperatures_c - reference_temperature_c))
}

simulate_storage_monthly <- function(
  factor_row,
  removal_schedule_name,
  removal_months,
  temperatures_c
) {
  ct_basis <- ct_basis_for_climate(qc_climate_zone)
  ct_value <- if (identical(ct_basis, "dry")) factor_row$CT_dry else factor_row$CT_wet

  monthly_ef <- calculate_monthly_ef(
    ef_ref = factor_row$EFs_TAN,
    ct = ct_value,
    temperatures_c = temperatures_c
  )

  stored_tan <- 0
  cumulative_nh3 <- 0
  rows <- vector("list", total_months)

  for (period_index in seq_len(total_months)) {
    month_index <- ((period_index - 1) %% length(month_labels)) + 1
    year_index <- ((period_index - 1) %/% length(month_labels)) + 1

    tan_start <- stored_tan
    tan_removed <- if (month_index %in% removal_months) {
      tan_start * removal_efficiency
    } else {
      0
    }

    tan_after_removal <- tan_start - tan_removed
    available_tan <- tan_after_removal + monthly_tan_kg
    ef_this_month <- monthly_ef[month_index]
    nh3_loss <- min(available_tan, available_tan * ef_this_month)
    monthly_ef_tan_stored <- ifelse(
      available_tan > 0,
      nh3_loss / available_tan,
      NA_real_
    )
    stored_tan <- available_tan - nh3_loss
    cumulative_nh3 <- cumulative_nh3 + nh3_loss

    rows[[period_index]] <- data.frame(
      Animal_Type = factor_row$Animal_Type,
      Storage = factor_row$Storage,
      Manure_type = factor_row$Manure_type,
      Combination = paste(factor_row$Animal_Type, factor_row$Storage, factor_row$Manure_type, sep = " | "),
      Climate_zone = qc_climate_zone,
      Removal_Schedule = removal_schedule_name,
      Removal_Months = paste(month_labels[removal_months], collapse = ";"),
      Year = year_index,
      Month = month_labels[month_index],
      Month_Index = month_index,
      Period = period_index,
      Temperature_C = temperatures_c[month_index],
      CT_Basis = ifelse(identical(ct_basis, "dry"), "Dry", "Wet"),
      CT = ct_value,
      EFs_TAN = factor_row$EFs_TAN,
      Corrected_EF_TAN = ef_this_month,
      TAN_Start_Stored_kg = tan_start,
      TAN_Removed_kg = tan_removed,
      TAN_After_Removal_kg = tan_after_removal,
      TAN_Input_kg = monthly_tan_kg,
      TAN_Available_kg = available_tan,
      NH3_N_kg = nh3_loss,
      Monthly_EF_TAN_Stored = monthly_ef_tan_stored,
      Cumulative_NH3_N_kg = cumulative_nh3,
      TAN_End_Stored_kg = stored_tan,
      stringsAsFactors = FALSE
    )
  }

  do.call(rbind, rows)
}
